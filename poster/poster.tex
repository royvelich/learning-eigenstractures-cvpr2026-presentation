\begin{block}{\LARGE TL;DR}
  \LARGE
  A neural network learns the Laplacian eigenbasis straight from a point
  cloud --- no mesh, no operator, no eigensolve. The whole pipeline is
  end-to-end differentiable, mesh-free, and plugs into any learning
  setup. The same recipe extends to other operators (Schr\"odinger,
  Hadamard) and to image-feature manifolds.
\end{block}

  \secTitle{cTitlePlum}{The optimality theorem \;{\normalsize\normalfont(Aflalo et al., 2016)}}
  \vspace{0.8ex}
  % Local box style for this subsection.
  {\setlength{\fboxsep}{0.7em}\setlength{\fboxrule}{0.6pt}%
  \fcolorbox{cBrandTo!35!white}{cBrandTo!8!white}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
      \Large\itshape\centering
      Let $L \succeq 0$ be the Laplacian,
      $\mathcal{C}_L = \{\, f : \langle f, L f\rangle \le 1 \,\}$,
      and $p_L = \mathrm{Uniform}(\mathcal{C}_L)$.
    \end{minipage}%
  }\par
  \vspace{1.2ex}
  \fcolorbox{cBrandTo!35!white}{cBrandTo!8!white}{%
    \begin{minipage}[t]{\dimexpr0.555\linewidth-2\fboxsep-2\fboxrule\relax}
      \Large\justifying
      \noindent\textbf{Theorem 1.}\;
      \emph{Among all orthonormal $k$-subspaces, the expected
            squared reconstruction error under $p_L$ is minimised by
            the first $k$ eigenvectors of $L$:}
      \[
        \arg\!\min_{\{b_i\}_{i=1}^{k}}\;
          \mathbb{E}_{f \sim p_L}\, \bigl\|f - \Pi_k f\bigr\|^{2}
        \;=\; \{\varphi_1, \dots, \varphi_k\}.
      \]
    \end{minipage}%
  }\hfill
  \fcolorbox{cBrandTo!35!white}{cBrandTo!8!white}{%
    \begin{minipage}[t]{\dimexpr0.415\linewidth-2\fboxsep-2\fboxrule\relax}
      \Large\justifying
      \noindent\textbf{Theorem 2.}\;
      \emph{Under that basis the worst-case reconstruction error is
            controlled by the next eigenvalue:}
      \[
        \max_{f \in \mathcal{C}_L}\; \bigl\|f - \Pi_k f\bigr\|^{2}
        \;=\; \tfrac{1}{\lambda_{k+1}}.
      \]
    \end{minipage}%
  }\par
